$ErrorActionPreference = "Stop"
function Get-NL([string]$c) { if ($c.Contains("`r`n")) { "`r`n" } else { "`n" } }
function Ensure-Replace([string]$c, [string]$oldCRLF, [string]$newCRLF, [string]$label) {
  if ($c.Contains($newCRLF.Replace("`r`n","`n")) -or $c.Contains($newCRLF)) {
    Write-Host "SKIP already: $label"
    return $c
  }
  $nl = Get-NL $c
  $old = $oldCRLF -replace "`r`n", $nl
  $new = $newCRLF -replace "`r`n", $nl
  if (-not $c.Contains($old)) {
    throw "FAIL marker not found: $label"
  }
  Write-Host "OK replace: $label"
  return $c.Replace($old, $new)
}
function Write-FileUtf8([string]$path, [string]$c) {
  $utf8 = New-Object System.Text.UTF8Encoding $false
  [IO.File]::WriteAllText((Resolve-Path $path), $c, $utf8)
}
